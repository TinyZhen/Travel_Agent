% Shared style of every Roameo figure (the deck, the poster and the report).
% UMass Boston palette, kept calm. Type is near-black on white. UMB blue carries the structure
% (axes, strokes, the three idea marks). One warm accent, UMB orange, marks only the moment of
% tension (the $180 gap). Green marks only a passed test or an approved plan. No decorative rules,
% no dashed boxes, no shadows, no gradients.
\usepackage[T1]{fontenc}
\usepackage[scaled]{helvet}
\renewcommand{\familydefault}{\sfdefault}
\usepackage{tikz}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usepackage{fontawesome5}
\usepackage{amsmath}
\hyphenpenalty=10000
\exhyphenpenalty=10000
\usetikzlibrary{arrows.meta,positioning,calc,fit,backgrounds,shapes.misc}

% Colours
\definecolor{ink}{HTML}{1A1A1A}       % all type
\definecolor{umb}{HTML}{005A8B}       % UMB blue, the structure
\definecolor{umbsoft}{HTML}{E9F2F8}   % its tint, for panel backgrounds
\definecolor{primary}{HTML}{08457E}   % slide titles
\definecolor{navy}{HTML}{132245}      % UMB dark navy, title bands only
\definecolor{accent}{HTML}{FF791F}    % UMB orange, tension only (fills and strokes)
\definecolor{accentink}{HTML}{C4540A} % the same orange, darker, for text on white
\definecolor{accentsoft}{HTML}{FFF1E6}
\definecolor{ok}{HTML}{2E7D32}        % a passed test or an approved plan
\definecolor{oksoft}{HTML}{E8F3E9}
\definecolor{muted}{HTML}{5F6368}
\definecolor{quiet}{HTML}{9A9A9A}
\definecolor{rulegrey}{HTML}{C9CED6}
\definecolor{gridgrey}{HTML}{E4E7EB}
\definecolor{faint}{HTML}{F4F5F7}

% The three idea marks, the same in the deck, the poster and the report.
\newcommand{\markagents}{\faSearch}
\newcommand{\markcode}{\ensuremath{\boldsymbol{\mathsf{\Sigma}}}}

\tikzset{
  >={Stealth[length=2.4mm,width=1.8mm]},
  every picture/.style={line cap=butt, line join=miter},
  hd/.style={font=\bfseries, text=ink, align=left},
  sub/.style={text=muted, align=left},
  panel/.style={fill=umbsoft, draw=none, rounded corners=3pt, inner sep=6pt, align=left, text=ink},
  card/.style={fill=faint, draw=none, rounded corners=3pt, inner sep=5pt, align=left, text=ink},
  sys/.style={draw=umb, line width=0.8pt, fill=white, rounded corners=3pt, inner sep=5pt, align=center, text=ink},
  sysfill/.style={draw=none, fill=umb, rounded corners=3pt, inner sep=5pt, align=center, text=white},
  code/.style={draw=ink, line width=0.8pt, fill=white, rounded corners=3pt, inner sep=5pt, align=center, text=ink},
  warm/.style={draw=accent, line width=1pt, fill=accentsoft, rounded corners=3pt, inner sep=5pt, align=left, text=ink},
  receipt/.style={draw=ok, line width=1pt, fill=oksoft, rounded corners=3pt, inner sep=5pt, align=left, text=ink},
  mark/.style={circle, fill=umb, text=white, inner sep=0pt, minimum size=7mm, font=\bfseries},
  link/.style={->, draw=muted, line width=0.7pt},

  oklink/.style={->, draw=ok, line width=1pt},
  warmlink/.style={->, draw=accent, line width=1pt},
}

% Charts: a real axis, few light gridlines, no frame on the top and right.
\pgfplotsset{
  roameo/.style={
    axis line style={draw=muted, line width=0.5pt},
    tick style={draw=muted, line width=0.5pt},
    major tick length=3pt,
    tick label style={text=ink, /pgf/number format/assume math mode=true},
    label style={text=ink},
    title style={font=\bfseries, text=ink},
    grid style={draw=gridgrey, line width=0.4pt},
    every axis plot/.append style={line width=0.6pt},
    legend style={draw=none, fill=none, text=ink},
    legend cell align=left,
  },
}
